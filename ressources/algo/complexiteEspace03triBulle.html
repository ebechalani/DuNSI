from random import randint

liste = []
for i in range(0, 30):
    liste.append(randint(1,100))

def trierABulle(liste):
    inversion = True
    indiceMax = len(liste)-1
    indiceDebut = 0
    while (inversion == True):
        inversion = False
        i = indiceDebut
        if i == -1:
            i = 0
        while (i < indiceMax):
            if (liste[i] > liste[i+1]):
                liste[i:i+2] = [liste[i+1],liste[i]]
                if inversion == False:
                    inversion = True
                    indiceDebut = i-1
            i = i + 1
        indiceMax = indiceMax - 1

print(liste)
trierABulle(liste)
print(liste)
