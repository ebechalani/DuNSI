def maximum(liste):
    assert len(liste) > 0
    max = liste[0]
    for i in range(0, len(liste)):
        if liste[i] > max:
            max = liste[i]
        #Invariant de boucle : max == maximum(liste[0:i+1])
        # variant de boucle : len(liste)-1 - i
    assert max == maximum(liste) # et maximum(liste) = maximum(liste[0:len(liste)+1])
    return max
