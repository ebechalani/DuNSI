class Calculatrice:
    def __init__(self):
        self.pile = []

    def empiler(self, nombre):
        assert type(nombre) == int or type(nombre) == float
        self.pile.append(nombre)

    def ajouter(self):
        assert len(self.pile) >= 2, "2 arguments attendus pour l'addition"
        operande2 = self.pile.pop()
        operande1 = self.pile.pop()
        resultat = operande1 + operande2
        self.pile.append(resultat)

    def sommet(self):
        return self.pile[-1]

    def estVide(self):
        return len(self.pile) == 0
